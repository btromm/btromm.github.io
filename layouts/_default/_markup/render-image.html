{{/* layouts/_default/_markup/render-image.html Based on the typo theme's render hook (photo-inline wrapper, --w/--h aspect-ratio variables, figure + caption, #tag -> img-tag classes). On pages with layout "photos", PNG/JPEG page-bundle images are served as resized WebP with a srcset instead of the full-size original. Everywhere else, images render exactly as the theme did. */}} {{- /* 1. Resolve the image resource: page bundle -> parent bundle -> assets/ */ -}} {{- $u := urls.Parse .Destination -}} {{- $img := false -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $img = .PageInner.Resources.Get $path -}} {{- if and (not $img) .PageInner.Parent -}} {{- $img = .PageInner.Parent.Resources.Get $path -}} {{- end -}} {{- if not $img -}} {{- $img = resources.Get $path -}} {{- end -}} {{- end -}} {{- /* 2. Tag classes from the URL fragment: photo.png#wide#dark -> "img-wide img-dark" */ -}} {{- $classes := "" -}} {{- range split $u.Fragment "#" -}} {{- with . -}}{{- $classes = printf "%s img-%s" $classes . -}}{{- end -}} {{- end -}} {{- $isPhotos := eq (.Page.Param "layout") "photos" -}} {{- $text := .Text -}} {{- /* 3. Build responsive WebP variants (photos pages, PNG/JPEG only). Never upscale; the largest variant is capped at 2000px wide. */ -}} {{- $srcset := slice -}} {{- $src := "" -}} {{- if and $isPhotos $img -}} {{- if in (slice "png" "jpeg") $img.MediaType.SubType -}} {{- $base := $img -}} {{- if eq $img.MediaType.SubType "jpeg" -}} {{- $base = $img.Filter images.AutoOrient -}} {{- end -}} {{- range slice 600 1200 2000 -}} {{- if lt . $base.Width -}} {{- $v := $base.Resize (printf "%dx webp q75" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $v.RelPermalink $v.Width) -}} {{- if eq . 1200 -}}{{- $src = $v.RelPermalink -}}{{- end -}} {{- end -}} {{- end -}} {{- if le $base.Width 2000 -}} {{- /* Original is <= 2000px: add a full-width WebP (converted, not enlarged) */ -}} {{- $v := $base.Process "webp q75" -}} {{- $srcset = $srcset | append (printf "%s %dw" $v.RelPermalink $v.Width) -}} {{- if not $src -}}{{- $src = $v.RelPermalink -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* 4. URL for anything not converted above (SVG, GIF, external, non-photos pages). Only computed when needed, so unused originals are not published. */ -}} {{- $url := $u.String -}} {{- if and $img (not $src) -}} {{- $url = $img.RelPermalink -}} {{- with $u.RawQuery -}}{{- $url = printf "%s?%s" $url . -}}{{- end -}} {{- end -}} {{- /* 5. Render */ -}} {{- if $isPhotos -}} {{- with $img }} {{- if $src }} {{- else if eq .MediaType.SubType "svg" }} {{- else }} {{- end }} {{- with $.Title }} {{ . | markdownify }} {{- end }} {{- end -}} {{- else -}} {{- end -}}
